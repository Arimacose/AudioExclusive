cmake_minimum_required(VERSION 3.24)
project(AudioExclusiveSetup LANGUAGES CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Build the Windows x64 installer")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
configure_file(Manage-AudioExclusive.ps1 "${CMAKE_CURRENT_BINARY_DIR}/Manage-AudioExclusive.ps1" COPYONLY)
configure_file(generated/payload.zip "${CMAKE_CURRENT_BINARY_DIR}/payload.zip" COPYONLY)
configure_file(generated/installer.rc "${CMAKE_CURRENT_BINARY_DIR}/installer.rc" @ONLY)
add_executable(audioexclusive_setup WIN32 Bootstrap.cpp "${CMAKE_CURRENT_BINARY_DIR}/installer.rc")
target_include_directories(audioexclusive_setup PRIVATE generated)
target_compile_definitions(audioexclusive_setup PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
target_compile_options(audioexclusive_setup PRIVATE /W4 /EHsc /utf-8 /guard:cf)
target_link_options(audioexclusive_setup PRIVATE /guard:cf /DYNAMICBASE /NXCOMPAT /HIGHENTROPYVA)
target_link_libraries(audioexclusive_setup PRIVATE bcrypt ole32 shell32)
set_target_properties(audioexclusive_setup PROPERTIES OUTPUT_NAME "AudioExclusive-Setup")
