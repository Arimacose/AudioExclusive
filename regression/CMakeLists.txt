cmake_minimum_required(VERSION 3.24)
project(AudioExclusiveMemoryFixTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(SRC "${CMAKE_CURRENT_LIST_DIR}/../backend-source")
enable_testing()
add_subdirectory("${SRC}/tests" upstream)
add_executable(cef_lifetime CefLifetimeTests.cpp)
target_include_directories(cef_lifetime PRIVATE "${SRC}/netease-source/src" "${SRC}/netease-source/third_party/cef" "${SRC}/netease-source/third_party/minhook/include")
target_compile_definitions(cef_lifetime PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN UNICODE _UNICODE)

add_test(NAME cef_lifetime COMMAND cef_lifetime)
