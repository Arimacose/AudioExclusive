cmake_minimum_required(VERSION 3.24)
project(NCExclusive VERSION 0.1.0 LANGUAGES C CXX RC)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "The pinned NetEase profile requires Windows x64")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_subdirectory(third_party/minhook EXCLUDE_FROM_ALL)
configure_file(ui/settings-exclusive.js "${CMAKE_CURRENT_BINARY_DIR}/settings-exclusive.js" COPYONLY)
configure_file(resources/ui.rc.in "${CMAKE_CURRENT_BINARY_DIR}/ui.rc" @ONLY)
add_library(nc_exclusive_hook SHARED src/Plugin.cpp src/NativeBackend.cpp src/ActualAudio.cpp src/AsioHost.cpp src/WasapiHost.cpp src/CefBridge.cpp
    src/Support.cpp "${CMAKE_CURRENT_BINARY_DIR}/ui.rc")
target_include_directories(nc_exclusive_hook PRIVATE src third_party/cef third_party/asio)
target_compile_definitions(nc_exclusive_hook PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(nc_exclusive_hook PRIVATE /W4 /EHsc /permissive- /utf-8 /guard:cf)
target_link_libraries(nc_exclusive_hook PRIVATE minhook bcrypt shlwapi ole32 uuid advapi32 user32)
target_link_options(nc_exclusive_hook PRIVATE /guard:cf)
set_target_properties(nc_exclusive_hook PROPERTIES OUTPUT_NAME "nc-exclusive-hook")
add_executable(nc_exclusive_launcher src/Launcher.cpp src/Support.cpp)
target_include_directories(nc_exclusive_launcher PRIVATE src)
target_compile_definitions(nc_exclusive_launcher PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(nc_exclusive_launcher PRIVATE /W4 /EHsc /permissive- /utf-8 /guard:cf)
target_link_options(nc_exclusive_launcher PRIVATE /guard:cf)
target_link_libraries(nc_exclusive_launcher PRIVATE bcrypt shlwapi)
set_target_properties(nc_exclusive_launcher PROPERTIES OUTPUT_NAME "nc-exclusive-launcher")
# No tests, autorun, install steps, or launch commands are attached to the build.
add_executable(nc_exclusive_probe src/Probe.cpp)
target_compile_definitions(nc_exclusive_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(nc_exclusive_probe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(nc_exclusive_probe PRIVATE ole32 uuid)
set_target_properties(nc_exclusive_probe PROPERTIES OUTPUT_NAME "nc-exclusive-probe")
install(TARGETS nc_exclusive_hook nc_exclusive_launcher RUNTIME DESTINATION . LIBRARY DESTINATION .)
add_executable(nc_exclusive_asio_probe src/AsioProbe.cpp src/AsioHost.cpp src/NativeBackend.cpp src/Support.cpp)
target_include_directories(nc_exclusive_asio_probe PRIVATE src third_party/asio third_party/minhook/include)
target_compile_definitions(nc_exclusive_asio_probe PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(nc_exclusive_asio_probe PRIVATE /W4 /EHsc /utf-8)
target_link_libraries(nc_exclusive_asio_probe PRIVATE ole32 uuid advapi32 user32 bcrypt shlwapi)
set_target_properties(nc_exclusive_asio_probe PROPERTIES OUTPUT_NAME "nc-exclusive-asio-probe")
